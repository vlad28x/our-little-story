it('rejects forged prefixes, incomplete choices, results and stale commands after the swap',()=>{
 const base=finishSky();const run=startRun(chapter4,base.campaign.records[3]!.story);
 expect(step(chapter4,run,{...command3(run),expectedTurn:run.turn-1}).kind).toBe('rejected');
 let s=reduceSave(reduceSave(base,{type:'unlock',code:chapters[3].code!}),{type:'start',chapter:4});
 while(s.campaign.active?.nodeId!=='c4.kitchen'){const r=s.campaign.active!;s=reduceSave(s,{type:'command',target:'campaign',command:command3(r)});}
 for(const change of [{teamName:'invented'},{wish:'adventure'},{kitchenResult:'feast'}]){const bad=structuredClone(s);Object.assign(bad.campaign.active!.story,change);expect(()=>read(bad)).toThrow();}

 expect(skyCommand(r).kind).toBe('advance');
});

it('migrates an actual old mock Telegram bank and writes the new format without repeating the reset',async()=>{
 const old=input.fixtures.ten,raw=JSON.stringify(old),data=new Map<string,string>();const chunks=raw.match(/[\s\S]{1,3000}/g)!;
 chunks.forEach((v,i)=>data.set(`five_meetings_v1_a_${i}`,v));const hash=createHash('sha256').update(raw).digest('hex');
 const head=JSON.stringify({version:1,current:{slot:'a',count:chunks.length,length:raw.length,hash},previous:null});data.set('five_meetings_v1_head',head);
 const api:CloudAPI={getItem:(k,done)=>{done(null,data.get(k)??'');},getItems:(ks,done)=>{done(null,Object.fromEntries(ks.map(k=>[k,data.get(k)??''])));},setItem:(k,v,done)=>{data.set(k,v);done(null,true);}};
 const cloud=new TelegramCloud(api),loaded=(await cloud.read()).save!;expect(loaded.campaign.completed).toHaveLength(10);expect(loaded.contentVersion).toBe(21);
 await cloud.write(loaded,head);expect((await new TelegramCloud(api).read()).save).toEqual(loaded);
});
