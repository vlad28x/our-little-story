it('dance and optional hoodie work for all eighth-chapter venues, gestures and endings',()=>{
 const base=finishR2(7);const cards=new Set<string>();
 for(const place of ['home','cafe'])for(const dance of ['close','lead','playful'])for(const tender of place==='home'?['curls','hair','cheek','neck','dance']:['curls','hair','cheek','dance'])for(const hoodie of place==='home'?['borrow','evening','none']:['none'])for(const ending of ['her','ask','near','tea']){
  const save=finishR2(8,{choices:{'c8.r2.place':place,'c8.r2.dance':dance,'c8.r2.tender':tender,'c8.r2.hoodie':hoodie,'c8.r2.ending':ending}},base);
  expect(parseSave(JSON.stringify(save))).toEqual(save);const record=save.campaign.records[8]!;
  expect(record.story.c8Dance).toBe(dance);expect(record.story.c8Tender).toBe(tender);cards.add(record.cardId);
  expect(record.story.c8Hoodie).toBe(place==='cafe'?undefined:hoodie==='borrow'?'borrowed':hoodie);
 }
 expect(cards.size).toBe(3);
},30000);
it('hoodie callbacks come only from canonical borrowed, and trial does not rewrite it',()=>{
 const base=finishR2(7);
 for(const hoodie of ['borrow','evening','none']){
  const eight=finishR2(8,{choices:{'c8.r2.place':'home','c8.r2.hoodie':hoodie}},base);
  for(const place of ['home','courtyard'])for(const response of ['return','keep']){
   const nine=finishR2(9,{choices:{'c9.r2.place':place,'c9.r2.hoodie':response}},eight);
   expect(parseSave(JSON.stringify(nine))).toEqual(nine);expect(nine.campaign.records[9]!.story.c9Hoodie).toBe(hoodie==='borrow'?(response==='return'?'returned':'kept'):undefined);
  }
  const trial=finishR2(8,{choices:{'c8.r2.place':'home','c8.r2.hoodie':hoodie==='borrow'?'none':'borrow'}},eight,true);
  expect(trial.campaign.records[8]).toEqual(eight.campaign.records[8]);
 }
});
it('all four final intentions work after a calm season, including first kiss and quiet overnight',()=>{
 const base=finishR2(9),cards=new Set<string>();
 for(const place of ['home','terrace'])for(const ending of ['goodnight','stay','private','overnight'])for(const affection of ['her','ask','quiet','leave'])for(const morning of ['hug','breakfast','coffee']){
  const p:R2Path={choices:{'c10.r2.place':place,'c10.r2.ending':ending,'c10.r2.move':'home','c10.r2.confirm.private':affection,'c10.r2.confirm.overnight':affection,'c10.r2.morning.choice':morning}};
  const save=finishR2(10,p,base);expect(parseSave(JSON.stringify(save))).toEqual(save);
  const record=save.campaign.records[10]!;cards.add(record.cardId);
  if(record.story.c10Ending==='private'||record.story.c10Ending==='overnight')expect(record.story.c10Venue).toBe('home');
  if(record.story.c10Ending==='overnight'){expect(record.story.c10Morning).toBe(morning);expect(cardBackground(save.collection.cards.find(c=>c.id==='c10.overnight')!)).toBe('bg.s2.kitchen.morning');}
 }
 expect(cards.size).toBe(4);
});
