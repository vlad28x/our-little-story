test('legacy upgrade preserves original bytes until explicit confirmation and keeps first-five records',async({page})=>{
 const old=oldEight as unknown as SaveV1;await openOffline(page,old,'/chapters');await expect(page.getByRole('heading',{name:'Сезон обновился'})).toBeVisible();expect(await page.evaluate(k=>localStorage.getItem(k),key)).toBe(JSON.stringify(old));
 await page.getByRole('button',{name:'Посмотреть первые пять встреч',exact:true}).click();await page.evaluate(()=>{location.hash='/settings';});
 const downloading=page.waitForEvent('download');await page.getByRole('button',{name:'Скачать копию прогресса',exact:true}).click();const download=await downloading;expect(await readFile((await download.path())!,'utf8')).toBe(JSON.stringify(old));
 await page.getByRole('button',{name:'К обновлению сезона',exact:true}).click();
 await page.getByRole('button',{name:'Начать второй сезон заново',exact:true}).click();
 await expect.poll(()=>page.evaluate(k=>JSON.parse(localStorage.getItem(k)!).contentVersion,key)).toBe(11);
 expect(await page.evaluate(k=>localStorage.getItem(k+':legacy-r1'),key)).toBe(JSON.stringify(old));const save=parseSave((await page.evaluate(k=>localStorage.getItem(k),key))!);expect(save.campaign.active?.chapter).toBe(6);
 for(const chapter of [1,2,3,4,5] as const)expect(save.campaign.records[chapter]).toEqual(old.campaign.records[chapter]);
});
test('Telegram legacy preflight waits for confirmation, then commits through checked cloud head',async({page})=>{
 const old=oldEight as unknown as SaveV1,data=legacyCloudData(old),head=data.five_meetings_v1_head;await openOffline(page,old,'/chapters',{seedBrowser:false,telegram:{id:42,data}});
 await expect(page.getByRole('heading',{name:'Сезон обновился'})).toBeVisible();await page.waitForTimeout(900);
 expect(await page.evaluate(()=>window.testCloud.data.five_meetings_v1_head)).toBe(head);
 await page.getByRole('button',{name:'Начать второй сезон заново',exact:true}).click();
 await expect.poll(()=>page.evaluate(()=>window.testCloud.data.five_meetings_v1_head)).not.toBe(head);
 const remote=await page.evaluate(()=>{const d=window.testCloud.data,h=JSON.parse(d.five_meetings_v1_head).current;return Array.from({length:h.count},(_,i)=>d[`five_meetings_v1_${h.slot}_${i}`]).join('');});
 expect(parseSave(remote).campaign.active?.chapter).toBe(6);
});
