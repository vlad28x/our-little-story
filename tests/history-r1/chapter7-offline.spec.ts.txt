const key='five-meetings:/:save';
async function finish(page:Page,p:SeventhPath=defaultSeventh){
 for(let i=0;i<200;i++){
  const state=await page.locator('.story-screen,.postcard').first().evaluate(el=>({finished:el.matches('.postcard'),id:el.getAttribute('data-node')}));if(state.finished)return;
  const id=state.id;
  const choices:Record<string,number>={'c7.guide':['angel','devil','none'].indexOf(p.guide),'c7.thought':['curious','playful','steady'].indexOf(p.thought),'c7.distance':['arm','near','space'].indexOf(p.distance),'c7.envelope.offer':p.bonus?0:1,'c7.ending':['open','promise','calm'].indexOf(p.ending)};
  if(id&&id in choices){await page.locator('.choices button').nth(choices[id]).click();continue;}
  if(id==='c7.note'){if(p.note){await page.getByLabel('Твой ответ',{exact:true}).fill(p.note);await page.getByRole('button',{name:'Сохранить для себя',exact:true}).click();}else await page.getByRole('button',{name:'Пропустить',exact:true}).click();continue;}
  if(id==='c7.council'){
   if(await page.locator('.council-answers').count()){
    const save=await page.evaluate(k=>JSON.parse(localStorage.getItem(k)!),key);const target=(await page.evaluate(()=>location.hash)).includes('mode=trial')?'trial':'campaign';
    const d=target==='trial'?save.trial.run.activity:save.campaign.active.activity;const answer=p.result.answers[d.answers.length];
    const ids=[['question','challenge','ordinary'],['lead','invite','stay'],['direct','playful','quiet']];
    if(answer)await page.locator('.council-answers button').nth(ids[d.answers.length].indexOf(answer)).click();else await page.getByRole('button',{name:'Продолжить прогулку без игры',exact:true}).click();
   }else await page.locator('.between-lines button').filter({hasText:/^(Дальше|Следующая ситуация|Завершить разговор|Вернуться к прогулке)/}).click();continue;
  }
  await page.getByRole('button',{name:'Дальше',exact:false}).click();
 }
 throw Error('Chapter seven did not finish');
}

test('chapter seven plays from its code on a 320px phone and preserves the first six chapters',async({page})=>{
 await page.setViewportSize({width:320,height:568});const prior=finishSix(),errors:string[]=[];page.on('pageerror',e=>errors.push(e.message));await openOffline(page,prior);
 await page.getByLabel('Код',{exact:true}).fill(seventhCode);await page.getByRole('button',{name:'Открыть встречу',exact:true}).click();await expect(page.locator('.chapter-list li')).toHaveCount(5);await page.getByRole('button',{name:'Начать встречу 7',exact:false}).click();
 await finish(page);await expect(page.getByRole('heading',{name:'Сказано вслух',exact:true})).toBeVisible();const save=parseSave(await page.evaluate(k=>localStorage.getItem(k)!,key));expect(save.campaign.completed).toEqual([1,2,3,4,5,6,7]);for(const id of [1,2,3,4,5,6] as const)expect(save.campaign.records[id]).toEqual(prior.campaign.records[id]);expect(save.privateNotes[6]).toEqual(prior.privateNotes[6]);expect(save.privateNotes[7]?.campaign?.text).toBe(defaultSeventh.note);expect(errors).toEqual([]);await page.screenshot({path:'outputs/CHAPTER_07_RESULT_320.png',fullPage:true});
});
