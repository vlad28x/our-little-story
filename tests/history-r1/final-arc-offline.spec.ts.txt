async function state(page:Page){return parseSave(await page.evaluate(k=>localStorage.getItem(k)!,key));}
async function finish(page:Page,p:ArcPath){
 for(let i=0;i<250;i++){
  const screen=await page.locator('.story-screen,.postcard').first().evaluate(el=>({end:el.matches('.postcard'),id:el.getAttribute('data-node')}));if(screen.end)return;
  const save=await state(page),trial=(await page.evaluate(()=>location.hash)).includes('mode=trial'),r=trial?save.trial!.run:save.campaign.active!;const a=actionArc(r,p,trial?'trial':'campaign');if(a.type!=='command')throw Error('Wrong action');const c=a.command;
  if(c.kind==='advance'){await page.getByRole('button',{name:'Дальше',exact:false}).click();continue;}
  if(c.kind==='choose'){const n=getChapter(r.chapter)!.nodes[r.nodeId];if(n.kind!=='choice')throw Error('Not a choice');await page.locator('.choices button').nth(n.choices.findIndex(v=>v.id===c.choiceId)).click();continue;}
  if(c.kind==='note-result'){if(c.status==='saved'){await page.getByLabel('Твой ответ',{exact:true}).fill(p.note!);await page.getByRole('button',{name:'Сохранить для себя',exact:true}).click();}else await page.getByRole('button',{name:'Пропустить',exact:true}).click();continue;}
  if(c.kind==='label-result'){if(c.status==='saved'){await page.getByLabel('Ваше маленькое слово',{exact:true}).fill(p.wordText!);await page.getByRole('button',{name:'Назвать так',exact:true}).click();}else await page.getByRole('button',{name:'Без названия',exact:true}).click();continue;}
  if(c.kind==='game-action'){
   let label='';if(c.game==='dessert')label=c.action.kind==='set-topping'?toppingNames[c.action.value]:c.action.kind==='set-mark'?markNames[c.action.value]:'Убрать украшение';
   if(c.game==='movie-setup')label=c.action.kind==='set-seat'?seatNames[c.action.value]:c.action.kind==='set-light'?lightNames[c.action.value]:c.action.kind==='set-drink'?drinkNames[c.action.value]:'Начать настройку заново';
   if(c.game==='evening-light')label=c.action.kind==='toggle'?'Переключить лампу':{warm:'Тёплый свет',dim:'Приглушить',unchanged:'Оставить как есть'}[c.action.level];
   await page.getByRole('button',{name:label,exact:false}).click();continue;
  }
  if(c.kind==='game-result'){
   const result=c.result;const label=result.kind==='dessert'?(result.mode==='completed'?'Готово':result.mode==='shared'?'Давай вместе':'Оставить десерт как есть'):result.kind==='movie-setup'?(result.mode==='completed'?'Готово':result.mode==='shared'?'Давай как ты подготовил':'Продолжить без настройки'):'Так хорошо';await page.getByRole('button',{name:label,exact:true}).click();continue;
  }
 }
 throw Error('Final arc did not finish');
}
